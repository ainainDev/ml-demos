🐍 The from-scratch class (Python)

import numpy as np

class KMeans:
    # n-dimensional Euclidean distance via dot product
    def _distance(self, a, b):
        diff = a - b
        return np.sqrt%(np.dot(diff, diff))

    def __init__(self, n_clusters=3, max_iterations=100):
        self.n_clusters = n_clusters
        self.max_iterations = max_iterations
        self.centroids = None

    # STEP 1: every point joins its nearest centroid
    def assign_clusters(self, X):
        labels = []
        for x in X:
            d = [self._distance(x, c) for c in self.centroids]
            labels.append(np.argmin(d))
        return np.array(labels)

    # STEP 2: each centroid moves to its cluster's mean
    def move_centroids(self, X, labels):
        return np.array([
            X[labels == j].mean(axis=0)
            for j in range(self.n_clusters)
        ])

    def fit(self, X):
        # random init: k random data points become centroids
        idx = np.random.choice(len(X), self.n_clusters, replace=False)
        self.centroids = X[idx].astype(float)

        for i in range(self.max_iterations):
            labels = self.assign_clusters(X)
            old = self.centroids.copy()
            self.centroids = self.move_centroids(X, labels)
            # convergence: stop when no centroid moved
            if np.all(old == self.centroids):
                break
        return labels
